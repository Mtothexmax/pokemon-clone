# walks on when you leave. The goal pill shows the quest step.

screen play
wait_for game_state.state.assets_pending == 0 15000

call life {"freeze": false, "pip_home": true}

call teleport {"x": 30, "z": 0}

expect ui.slots.goal contains Visit Prof. Quill
