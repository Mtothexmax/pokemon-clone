# See-through: a building between the camera and you fades out, the camera does not move.
# Run: node C:/dev/wasmOnly/tools/bridge.mjs run tests/fade.txt --port 9223 --url http://localhost:8091/
reload
screen play
wait_for game_state.state.assets_pending == 0 15000
call clock {"hours": 12}
resume
# out in the open: nothing hides you, nothing is faded
call teleport {"x": -10, "z": 0.5}
frames 60
expect game_state.state.hiding == 0
expect game_state.state.faded == 0
# just north of the lab: the camera looks over the lab at you, the lab fades
call teleport {"x": 13, "z": 4}
frames 60
expect game_state.state.hiding >= 1
expect game_state.state.faded >= 1
expect game_state.state.camera.2 > 15
# solid fade: the faded lab is drawn as its front surface only (no inner bricks / back walls showing)
expect stats.render.solid_fade_draws > 0
screenshot shots/fade-lab.png
# north of your house
call teleport {"x": -15, "z": -18}
frames 60
expect game_state.state.hiding >= 1
screenshot shots/fade-house.png
# back in the open: everything comes back
call teleport {"x": -10, "z": 0.5}
frames 90
expect game_state.state.faded == 0

call teleport {"x": -8, "z": 13}
frames 60
