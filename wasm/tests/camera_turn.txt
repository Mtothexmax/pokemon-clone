# On a PC the camera turns with the right mouse button and the keys walk the way it looks; with the button up the
# mouse does nothing. (Phones keep the fixed view: swipe.txt checks cam_yaw == 0 while swiping.)
# Run: node C:/dev/wasmOnly/tools/bridge.mjs run tests/camera_turn.txt --port 9223 --url http://localhost:8091/
reload
screen play
wait_for game_state.state.assets_pending == 0 15000
call save_reset
call clock {"hours": 12}
call life {"freeze": true, "pip_home": true}
resume
call teleport {"x": -10, "z": 0.5}
frames 20
expect game_state.state.cam_yaw == 0
# the mouse alone does not turn it
call input {"dx": 200, "dy": 0}
frames 3
expect game_state.state.cam_yaw == 0
# right button held: 314 px turn it a quarter (0.005 rad per px)
call input {"button": 2, "down": true}
frames 1
call input {"dx": -314, "dy": 0}
frames 3
call input {"button": 2, "down": false}
frames 3
expect game_state.state.cam_yaw ~= 1.57
