# The tall grass moves (grass.rs): the engine bends the tufts (Material::bend) away from the pushers the game sets every
# frame (you, villagers, Voxelmon) plus the engine's wind; no tuft transform is written (no instance upload for them).
# `near` / `bent` / `max_bend` are the engine's lean (common.wgsl push_offset) recomputed from the walkers.
# Run: node C:/dev/wasmOnly/tools/bridge.mjs run tests/grass.txt --port 9223 --url http://localhost:8091/
reload
screen play
wait_for game_state.state.assets_pending == 0 20000
call save_reset
call time {"hours": 12}
call life {"freeze": true, "pip_home": true}
call flags {"gate_open": true}
call vox {"rate": 0}
resume
call teleport {"x": 2.5, "z": -22}
frames 75
expect game_state.state.grass.tufts > 600
expect game_state.state.grass.patches.0 > 150
expect game_state.state.grass.bent == 0
expect game_state.state.grass.writes == 0
# the bending materials (one per green), the pushers the renderer has (you at least) and the wind
expect bend.materials.length >= 4
expect bend.materials.0.bend ~= 1.1
expect bend.pushers.length >= 1
expect bend.wind.2 > 0.15

down KeyW

expect stats.render.instance_upload_bytes < 60000
